// Get the metadata for the Account object
Schema.SObjectType accountType = Schema.getGlobalDescribe().get('Account');
Map<String, Schema.SObjectField> fieldMap = accountType.getDescribe().fields.getMap();

// List of specific Account fields to check
List<String> fieldsToCheck = new List<String>{'Name', 'Phone', 'AnnualRevenue', 'OwnerId'};

System.debug('--- PROOF: Checking isHistoryTracked() method ---');

for (String fieldName : fieldsToCheck) {
    if (fieldMap.containsKey(fieldName)) {
        // 1. Get the field token
        Schema.SObjectField fieldToken = fieldMap.get(fieldName);

        // 2. Get the field's DescribeResult
        Schema.DescribeFieldResult fieldDescribe = fieldToken.getDescribe();

        // 3. Call the isHistoryTracked() method and print the result
        // This line directly proves the method exists and works.
        Boolean isTracked = fieldDescribe.isHistoryTracked();
        
        System.debug('Field: ' + fieldDescribe.getLabel() + ' (' + fieldName + ') | Is History Tracked? -> ' + isTracked);
    }
}
System.debug('--- PROOF COMPLETE ---');
